class MatchEventViewModel : ViewModel() {

    private val _homeTeam = MutableLiveData<Team?>()
    val homeTeam: LiveData<Team?> = _homeTeam

    private val _awayTeam = MutableLiveData<Team?>()
    val awayTeam: LiveData<Team?> = _awayTeam

    private val _homeScore = MutableLiveData(0)
    val homeScore: LiveData<Int> = _homeScore

    private val _awayScore = MutableLiveData(0)
    val awayScore: LiveData<Int> = _awayScore

    private val _events = MutableLiveData<MutableList<MatchEvent>>(mutableListOf())
    val events: LiveData<MutableList<MatchEvent>> = _events

    private val yellowCardCount = mutableMapOf<String, Int>()
    private val redCardPlayers  = mutableSetOf<String>()

    private val _lastEvent = MutableLiveData<MatchEvent?>()
    val lastEvent: LiveData<MatchEvent?> = _lastEvent

    private val _pendingPKEvent = MutableLiveData<MatchEvent?>()
    val pendingPKEvent: LiveData<MatchEvent?> = _pendingPKEvent

    fun setTeams(home: Team, away: Team) {
        _homeTeam.value = home
        _awayTeam.value = away
    }

    fun reset() {
        _homeScore.value      = 0
        _awayScore.value      = 0
        _events.value         = mutableListOf()
        _lastEvent.value      = null
        _pendingPKEvent.value = null
        yellowCardCount.clear()
        redCardPlayers.clear()
    }

    fun addGoal(
        isHome: Boolean, scorerNumber: Int, assistNumber: Int?,
        isOwnGoal: Boolean, isPK: Boolean,
        minute: Int, period: Int, elapsedMillis: Long,
        gpsLocation: GpsPoint? = null
    ): MatchEvent? {
        val scoringTeam = if (isOwnGoal) {
            if (isHome) _awayTeam.value else _homeTeam.value
        } else {
            if (isHome) _homeTeam.value else _awayTeam.value
        }
        val team   = scoringTeam ?: return null
        val scorer = team.getPlayer(scorerNumber) ?: Player(scorerNumber, "Unknown")
        val assist = assistNumber?.let { team.getPlayer(it) ?: Player(it, "Unknown") }

        val event = MatchEvent(
            type = EventType.GOAL, team = team, player = scorer,
            assistPlayer = assist, minute = minute, period = period,
            elapsedMillis = elapsedMillis, isOwnGoal = isOwnGoal,
            isPK = isPK, gpsLocation = gpsLocation
        )
        if (isOwnGoal) {
            if (isHome) _awayScore.value = (_awayScore.value ?: 0) + 1
            else        _homeScore.value = (_homeScore.value ?: 0) + 1
        } else {
            if (isHome) _homeScore.value = (_homeScore.value ?: 0) + 1
            else        _awayScore.value = (_awayScore.value ?: 0) + 1
        }
        addEvent(event)
        return event
    }

    fun addFoul(
        isHome: Boolean, playerNumber: Int, foulType: FoulType,
        isPK: Boolean, minute: Int, period: Int, elapsedMillis: Long,
        gpsLocation: GpsPoint? = null
    ): MatchEvent? {
        val team   = (if (isHome) _homeTeam.value else _awayTeam.value) ?: return null
        val player = team.getPlayer(playerNumber) ?: Player(playerNumber, "Unknown")
        val event  = MatchEvent(
            type = EventType.FOUL, team = team, player = player,
            minute = minute, period = period, elapsedMillis = elapsedMillis,
            foulType = foulType, isPK = isPK, gpsLocation = gpsLocation
        )
        if (isPK) _pendingPKEvent.value = event
        else addEvent(event)
        return event
    }

    fun resolvePK(goalScored: Boolean, pkTakerNumber: Int? = null) {
        val foulEvent = _pendingPKEvent.value ?: return
        addEvent(foulEvent)
        _pendingPKEvent.value = null
        if (goalScored && pkTakerNumber != null) {
            val isHome = foulEvent.team == _homeTeam.value
            addGoal(
                isHome = !isHome, scorerNumber = pkTakerNumber,
                assistNumber = null, isOwnGoal = false, isPK = true,
                minute = foulEvent.minute, period = foulEvent.period,
                elapsedMillis = foulEvent.elapsedMillis,
                gpsLocation = foulEvent.gpsLocation
            )
        }
    }

    fun addCard(
        isHome: Boolean, playerNumber: Int, cardType: CardType,
        minute: Int, period: Int, elapsedMillis: Long,
        gpsLocation: GpsPoint? = null
    ): CardResult {
        val team   = (if (isHome) _homeTeam.value else _awayTeam.value) ?: return CardResult.ERROR
        val player = team.getPlayer(playerNumber) ?: Player(playerNumber, "Unknown")
        val key    = "${team.name}_$playerNumber"

        val resolvedCardType: CardType
        val result: CardResult

        when (cardType) {
            CardType.YELLOW -> {
                val count = (yellowCardCount[key] ?: 0) + 1
                yellowCardCount[key] = count
                if (count >= 2) {
                    redCardPlayers.add(key)
                    resolvedCardType = CardType.RED
                    result = CardResult.SECOND_YELLOW_RED
                } else {
                    resolvedCardType = CardType.YELLOW
                    result = CardResult.YELLOW
                }
            }
            CardType.RED -> {
                redCardPlayers.add(key)
                resolvedCardType = CardType.RED
                result = CardResult.RED
            }
        }

        val event = MatchEvent(
            type = EventType.CARD, team = team, player = player,
            minute = minute, period = period, elapsedMillis = elapsedMillis,
            cardType = resolvedCardType, gpsLocation = gpsLocation
        )
        addEvent(event)
        return result
    }

    fun hasRedCard(isHome: Boolean, playerNumber: Int): Boolean {
        val team = (if (isHome) _homeTeam.value else _awayTeam.value) ?: return false
        val key  = "${team.name}_$playerNumber"
        return redCardPlayers.contains(key) || (yellowCardCount[key] ?: 0) >= 2
    }

    fun getYellowCardCount(isHome: Boolean, playerNumber: Int): Int {
        val team = (if (isHome) _homeTeam.value else _awayTeam.value) ?: return 0
        return yellowCardCount["${team.name}_$playerNumber"] ?: 0
    }

    private fun addEvent(event: MatchEvent) {
        val current = _events.value ?: mutableListOf()
        current.add(event)
        _events.value  = current
        _lastEvent.value = event
    }

    fun getEventsByType(type: EventType): List<MatchEvent> =
        _events.value?.filter { it.type == type } ?: emptyList()

    fun getEventsByPeriod(period: Int): List<MatchEvent> =
        _events.value?.filter { it.period == period } ?: emptyList()

    }

    private fun setupCodeStep() {
        val yellowCodes = arrayOf("UB", "DT", "PI", "DR", "TR", "FPO")
        val redCodes = arrayOf("SFP", "VC", "S", "DOGSO", "AL", "2Y")

        binding.btnCycleCode.setOnClickListener {
            val codes = if (selectedCard == CardType.YELLOW) yellowCodes else redCodes

    // ── STEP 4: PACKAGE JSON & BEAM TO PHONE ──
    private fun commitEvent() {

        val period = timerViewModel.currentPeriod.value ?: 1
        val elapsed = timerViewModel.elapsedTimer.value ?: 0L

        // Local match data management fallback
        val teamName = if (isHome) timerViewModel.homeTeamName.value.orEmpty().ifBlank { "Home" } else timerViewModel.awayTeamName.value.orEmpty().ifBlank { "Away" }
        val team = if (isHome) eventViewModel.homeTeam.value else eventViewModel.awayTeam.value
        val playerName = team?.getPlayer(playerNumber)?.name ?: "Unknown"

        // Build a lean telemetry JSON to pass off structural tracking to the phone
        val eventJson = JSONObject().apply {
            put("id", UUID.randomUUID().toString())
            put("type", selectedType?.name)
            put("isHome", isHome)
            put("teamName", teamName)
            put("playerNumber", playerNumber)
            put("playerName", playerName)
            put("minute", minute)
            put("period", period)
            put("elapsedMillis", elapsed)
            if (selectedType == EventType.CARD) {
                put("cardType", selectedCard?.name)
                put("misconductCode", misconductCode)
            }
        }.toString()

        // Stream it immediately across the Data Layer
        CoroutineScope(Dispatchers.IO).launch {
            try {
                val request = PutDataMapRequest.create(DataLayerPaths.MSG_MATCH_EVENT).apply {
                    dataMap.putString(DataLayerPaths.KEY_EVENT_JSON, eventJson)
                    dataMap.putLong(DataLayerPaths.KEY_TIMESTAMP, System.currentTimeMillis())
                }
                Wearable.getDataClient(this@QuickEventActivity)
                    .putDataItem(request.asPutDataRequest().setUrgent())
                    .await()
            } catch (e: Exception) {
                android.util.Log.e("DATA_LAYER", "Failed to transfer event log telemetry", e)
            }
        }

    private fun computeAffineTransform(
        src: List<Pair<Float, Float>>,
        dst: List<Pair<Float, Float>>
    ): FloatArray {
        val n = src.size
        var sumX = 0f; var sumY = 0f; var sumU = 0f; var sumV = 0f
        var sumXX = 0f; var sumXY = 0f; var sumYY = 0f
        var sumXU = 0f; var sumYU = 0f; var sumXV = 0f; var sumYV = 0f

        for (i in 0 until n) {
            val x = src[i].first;  val y = src[i].second
            val u = dst[i].first;  val v = dst[i].second
            sumX+=x; sumY+=y; sumU+=u; sumV+=v
            sumXX+=x*x; sumXY+=x*y; sumYY+=y*y
            sumXU+=x*u; sumYU+=y*u; sumXV+=x*v; sumYV+=y*v
        }

        val det = sumXX * sumYY - sumXY * sumXY
        val a = if (det != 0f) (sumXU*sumYY - sumYU*sumXY) / det else 1f
        val b = if (det != 0f) (sumXX*sumYU - sumXY*sumXU) / det else 0f
        val c = (sumU - a*sumX - b*sumY) / n
        val d = if (det != 0f) (sumXV*sumYY - sumYV*sumXY) / det else 0f
        val e = if (det != 0f) (sumXX*sumYV - sumXY*sumXV) / det else 1f
        val f = (sumV - d*sumX - e*sumY) / n

        return floatArrayOf(a, b, c, d, e, f)
    }

                    DataLayerPaths.MSG_CALIBRATE_DONE

                    .await()
            } catch (e: Exception) { /* Phone unavailable */ }
        }

            "Calibration complete. ${calibratedPoints.size} points captured.",

    }

class WearDataService : WearableListenerService() {

    override fun onDataChanged(dataEvents: DataEventBuffer) {
        dataEvents.forEach { event ->
            if (event.type == DataEvent.TYPE_CHANGED) {
                val path    = event.dataItem.uri.path ?: return@forEach
                val dataMap = DataMapItem.fromDataItem(event.dataItem).dataMap
                when (path) {
                    DataLayerPaths.DATA_ROSTER_HOME,
                    DataLayerPaths.DATA_ROSTER_AWAY -> {
                        val rosterJson = dataMap.getString(DataLayerPaths.KEY_ROSTER_JSON) ?: return@forEach
                        val isHome     = path == DataLayerPaths.DATA_ROSTER_HOME
                        broadcastRosterUpdate(rosterJson, isHome)
                    }
                    DataLayerPaths.DATA_GPS_BATCH -> {
                        val driftLat = dataMap.getDouble(DataLayerPaths.KEY_DRIFT_LAT, 0.0)
                        val driftLng = dataMap.getDouble(DataLayerPaths.KEY_DRIFT_LNG, 0.0)
                        broadcastGpsCorrection(driftLat, driftLng)
                    }
                    DataLayerPaths.MSG_CALIBRATE_POINT -> {
                        val lat = dataMap.getDouble(DataLayerPaths.KEY_GPS_LAT, 0.0)
                        val lng = dataMap.getDouble(DataLayerPaths.KEY_GPS_LNG, 0.0)
                        val acc = dataMap.getFloat(DataLayerPaths.KEY_GPS_ACCURACY, 5f)
                        broadcastPhoneGpsPoint(GpsPoint(lat, lng, acc, false))
                    }
                }
            }
        }
    }

    override fun onMessageReceived(messageEvent: MessageEvent) {
        when (messageEvent.path) {
            DataLayerPaths.MSG_RESET        -> broadcastReset()
            DataLayerPaths.DATA_MATCH_CONFIG -> broadcastMatchConfig(String(messageEvent.data))
        }
    }

    private fun broadcastRosterUpdate(json: String, isHome: Boolean) {
        LocalBroadcastManager.getInstance(this).sendBroadcast(
            Intent(ACTION_ROSTER_UPDATE).apply {
                putExtra(EXTRA_ROSTER_JSON, json); putExtra(EXTRA_IS_HOME, isHome) })
    }
    private fun broadcastGpsCorrection(driftLat: Double, driftLng: Double) {
        LocalBroadcastManager.getInstance(this).sendBroadcast(

                putExtra(EXTRA_DRIFT_LAT, driftLat); putExtra(EXTRA_DRIFT_LNG, driftLng) })

                putExtra(EXTRA_GPS_ACC, gps.accuracy) })
    }

        LocalBroadcastManager.getInstance(this).sendBroadcast(Intent(ACTION_RESET))
    }
    private fun broadcastMatchConfig(json: String) {

            Intent(ACTION_MATCH_CONFIG).apply { putExtra(EXTRA_CONFIG_JSON, json) })
    }

package com.example.refereetimer

import android.app.Application

class TimerViewModel(application: Application) : AndroidViewModel(application) {

    // CHANGED: Replaced SETUP with READY for the Big Green Button layout
    enum class TimerState { READY, IDLE, RUNNING, PAUSED, TIME_UP, BREAK, MATCH_OVER }

    private val _timerState   = MutableLiveData(TimerState.READY)
    val timerState: LiveData<TimerState> = _timerState

    private val _mainTimer    = MutableLiveData(0L)
    val mainTimer: LiveData<Long> = _mainTimer

    private val _elapsedTimer = MutableLiveData(0L)
    val elapsedTimer: LiveData<Long> = _elapsedTimer

    private val _stoppageTimer = MutableLiveData(0L)
    val stoppageTimer: LiveData<Long> = _stoppageTimer

    private val _currentPeriod = MutableLiveData(1)
    val currentPeriod: LiveData<Int> = _currentPeriod

    private val _totalPeriods  = MutableLiveData(3)
    val totalPeriods: LiveData<Int> = _totalPeriods

    private val _homeTeamName  = MutableLiveData("")
    val homeTeamName: LiveData<String> = _homeTeamName

    private val _awayTeamName  = MutableLiveData("")
    val awayTeamName: LiveData<String> = _awayTeamName

    // CHANGED: Default match parameters so the watch functions perfectly without a phone sync
    var regularPeriodsCount    = 2
    private var regularDurationMillis  = 45 * 60 * 1000L // 45 Minutes
    private var otDurationMillis       = 10 * 60 * 1000L // 10 Minutes
    private var breakDurationMillis    = 15 * 60 * 1000L // 15 Minutes
    private var hasOvertime            = false
    private var remainingMainTimeMillis = 45 * 60 * 1000L

    private var mainCountdownTimer: CountDownTimer? = null
    private var elapsedHandler  = Handler(Looper.getMainLooper())
    private var stoppageHandler = Handler(Looper.getMainLooper())

    private val elapsedRunnable = object : Runnable {
        override fun run() {
            _elapsedTimer.value = (_elapsedTimer.value ?: 0) + 1000L
            elapsedHandler.postDelayed(this, 1000L)
        }
    }

    private val stoppageRunnable = object : Runnable {
        override fun run() {
            _stoppageTimer.value = (_stoppageTimer.value ?: 0) + 1000L
            stoppageHandler.postDelayed(this, 1000L)
        }
    }

    fun setMatchParameters(
        regMins: Int, regSecs: Int, periods: Int,
        breakMins: Int, isOT: Boolean, otMins: Int, otSecs: Int,
        homeName: String, awayName: String
    ) {
        regularDurationMillis = (regMins * 60 * 1000L) + (regSecs * 1000L)
        breakDurationMillis   = breakMins * 60 * 1000L
        regularPeriodsCount   = periods
        hasOvertime           = isOT
        _totalPeriods.value   = if (hasOvertime) regularPeriodsCount + 2 else regularPeriodsCount
        otDurationMillis      = (otMins * 60 * 1000L) + (otSecs * 1000L)
        _homeTeamName.value   = homeName
        _awayTeamName.value   = awayName
        resetMatch()
    }

    fun resetMatch() {
        mainCountdownTimer?.cancel()
        elapsedHandler.removeCallbacks(elapsedRunnable)
        stoppageHandler.removeCallbacks(stoppageRunnable)
        _currentPeriod.value        = 1
        remainingMainTimeMillis     = regularDurationMillis
        _mainTimer.value            = regularDurationMillis
        _elapsedTimer.value         = 0L
        _stoppageTimer.value        = 0L
        _timerState.value           = TimerState.READY // Returns to the Big Green Button

        broadcastStateToPhone(TimerState.READY, regularDurationMillis)
    }

    fun onMainButtonClick() {
        when (_timerState.value) {
            TimerState.READY   -> {
                // Moving from the big green button to the active timer view
                _timerState.value = TimerState.IDLE
                broadcastStateToPhone(TimerState.IDLE, remainingMainTimeMillis)
            }
            TimerState.IDLE    -> startTimers()
            TimerState.RUNNING -> pauseTimers()
            TimerState.PAUSED  -> resumeTimers()
            TimerState.TIME_UP -> endPeriod()
            TimerState.BREAK   -> startNextPeriod()
            else -> {}
        }
    }

    private fun startTimers() {
        mainCountdownTimer = object : CountDownTimer(remainingMainTimeMillis, 1000) {
            override fun onTick(millisUntilFinished: Long) {
                _mainTimer.value        = millisUntilFinished
                remainingMainTimeMillis = millisUntilFinished
                broadcastStateToPhone(_timerState.value ?: TimerState.RUNNING, millisUntilFinished)
            }
            override fun onFinish() {
                // Wait for the referee's whistle.
                _mainTimer.value = 0L
                _timerState.value = TimerState.TIME_UP
                broadcastStateToPhone(TimerState.TIME_UP, 0L)
            }
        }.start()

        // Ensure elapsed time is running
        if (_timerState.value == TimerState.IDLE || _timerState.value == TimerState.BREAK) {
            elapsedHandler.post(elapsedRunnable)
        }
        _timerState.value = TimerState.RUNNING
        broadcastStateToPhone(TimerState.RUNNING, remainingMainTimeMillis)
    }

    private fun pauseTimers() {
        mainCountdownTimer?.cancel()
        stoppageHandler.post(stoppageRunnable)
        _timerState.value = TimerState.PAUSED
        broadcastStateToPhone(TimerState.PAUSED, remainingMainTimeMillis)
    }

    private fun resumeTimers() {
        stoppageHandler.removeCallbacks(stoppageRunnable)
        startTimers()
    }

    private fun endPeriod() {
        // Freeze the elapsed timer exactly here to preserve total playing time
        elapsedHandler.removeCallbacks(elapsedRunnable)

        val current = _currentPeriod.value ?: 1
        if (current < regularPeriodsCount) {
            // Move to halftime
            _timerState.value = TimerState.BREAK
            _mainTimer.value  = breakDurationMillis
            broadcastStateToPhone(TimerState.BREAK, breakDurationMillis)

            // Start the halftime countdown automatically
            mainCountdownTimer = object : CountDownTimer(breakDurationMillis, 1000) {
                override fun onTick(ms: Long) {
                    _mainTimer.value = ms
                    broadcastStateToPhone(TimerState.BREAK, ms)
                }
                override fun onFinish() { _mainTimer.value = 0L }
            }.start()
        } else {
            // End of regulation match
            _timerState.value = TimerState.MATCH_OVER
            broadcastStateToPhone(TimerState.MATCH_OVER, 0L)
        }
    }

    private fun startNextPeriod() {
        mainCountdownTimer?.cancel()
        val nextPeriod = (_currentPeriod.value ?: 1) + 1
        _currentPeriod.value        = nextPeriod
        remainingMainTimeMillis     = if (nextPeriod > regularPeriodsCount) otDurationMillis else regularDurationMillis

        // Clear stoppage, but leave elapsedTimer alone so total match time continues to accumulate
        _stoppageTimer.value        = 0L

        startTimers()
    }

    private fun broadcastStateToPhone(state: TimerState, remaining: Long) {
        val message = "STATE=$state|TIME=$remaining"
        CoroutineScope(Dispatchers.IO).launch {
            try {
                val nodes = Wearable.getNodeClient(getApplication<Application>()).connectedNodes.await()
                for (node in nodes as List<Node>) {
                    Wearable.getMessageClient(getApplication<Application>())
                        .sendMessage(node.id, "/match/sync", message.toByteArray())
                        .await()
                }
            } catch (e: Exception) {
                android.util.Log.e("SYNC_ERROR", "Data stream failed to synchronize with base engine", e)
            }
        }
    }
