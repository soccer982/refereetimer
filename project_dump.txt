data class MatchEvent(
    val id: String = UUID.randomUUID().toString(),
    val type: EventType,
    val team: Team,
    val player: Player,
    val playerOff: Player? = null,
    val assistPlayer: Player? = null,
    val minute: Int,
    val period: Int,
    val elapsedMillis: Long,
    val foulType: FoulType? = null,
    val cardType: CardType? = null,
    val misconductCode: String? = null,
    val isOwnGoal: Boolean = false,
    val isPK: Boolean = false,
    // NEW: Fields specifically for Kicks from the Penalty Mark (KFTM)
    val kftmResult: KftmResult? = null,
    val gpsLocation: GpsPoint? = null,
    val timestamp: Long = System.currentTimeMillis()
) {
    // NEW: Added KFTM_KICK to isolate shootout events from regular match events
    enum class EventType { GOAL, FOUL, CARD, SUBSTITUTION, KFTM_KICK }
    enum class FoulType  { DFK, IFK }
    enum class CardType  { YELLOW, RED }

    // NEW: Outcomes for a KFTM attempt
    enum class KftmResult { SCORED, MISSED, SAVED }

    fun toDisplayString(): String = when (type) {
        EventType.GOAL -> buildString {
            append("⚽ ${minute}' ")
            if (isOwnGoal) append("OG ")
            append("${team.name} #${player.jerseyNumber}")
            assistPlayer?.let { append(" (Assist: #${it.jerseyNumber})") }
            if (isPK) append(" [PK]")
        }
        EventType.FOUL -> buildString {
            append("🚩 ${minute}' ")
            append("${team.name} #${player.jerseyNumber} ")
            append(foulType?.name ?: "")
            if (isPK) append(" → PK")
        }
        EventType.CARD -> buildString {
            val icon = if (cardType == CardType.RED) "🟥" else "🟨"
            append("$icon ${minute}' ")
            append("${team.name} #${player.jerseyNumber}")
            if (!misconductCode.isNullOrBlank()) {
                append(" [$misconductCode]")
            }
        }
        EventType.SUBSTITUTION -> buildString {
            append("🔄 ${minute}' ")
            append("${team.name}: IN #${player.jerseyNumber} | OUT #${playerOff?.jerseyNumber ?: "?"}")
        }
        // NEW: Formats the shootout log for the phone UI
        EventType.KFTM_KICK -> buildString {
            val icon = if (kftmResult == KftmResult.SCORED) "✅" else "❌"
            append("$icon KFTM: ${team.name} #${player.jerseyNumber} - ${kftmResult?.name}")
        }
    }

package com.example.refereetimer.shared.models

data class Team(
    val name: String,
    val players: List<Player>,
    val coaches: List<String>,
    val isHome: Boolean,
    val jerseyColor: String = ""
) {
    val yellowCards: MutableMap<Int, Int> = mutableMapOf()
    val redCards: MutableSet<Int> = mutableSetOf()

    fun getPlayer(jerseyNumber: Int): Player? =
        players.find { it.jerseyNumber == jerseyNumber }

    fun addYellowCard(jerseyNumber: Int): CardResult {
        val current = yellowCards.getOrDefault(jerseyNumber, 0) + 1
        yellowCards[jerseyNumber] = current
        return if (current >= 2) CardResult.SECOND_YELLOW_RED
        else CardResult.YELLOW
    }

    fun addRedCard(jerseyNumber: Int): CardResult {
        redCards.add(jerseyNumber)
        return CardResult.RED
    }

    fun hasRedCard(jerseyNumber: Int): Boolean =
        redCards.contains(jerseyNumber) ||
        (yellowCards.getOrDefault(jerseyNumber, 0) >= 2)

    enum class CardResult {
        YELLOW,
        SECOND_YELLOW_RED,
        RED
    }
}

class MatchConfigActivity : AppCompatActivity() {
    private lateinit var binding: ActivityMatchConfigBinding

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        binding = ActivityMatchConfigBinding.inflate(layoutInflater)
        setContentView(binding.root)

        // Load existing preferences if they exist
        loadCurrentSettings()

        binding.btnSaveAndNext.setOnClickListener {
            saveSettings()
            // Proceed to Step 2: Roster Setup
            startActivity(Intent(this, RosterSetupActivity::class.java))
            finish() // Close this activity so the back button behaves cleanly
        }
    }

    private fun loadCurrentSettings() {
        val prefs = getSharedPreferences("MatchConfigPrefs", Context.MODE_PRIVATE)
        binding.inputPeriods.setText(prefs.getInt("periods", 2).toString())
        binding.inputDuration.setText(prefs.getInt("duration", 45).toString())
        binding.inputBreak.setText(prefs.getInt("breakDuration", 10).toString())
        binding.switchDrinkBreak.isChecked = prefs.getBoolean("drinkBreak", false)
    }

    private fun saveSettings() {
        val periods = binding.inputPeriods.text.toString().toIntOrNull() ?: 2
        val duration = binding.inputDuration.text.toString().toIntOrNull() ?: 45
        val breakDuration = binding.inputBreak.text.toString().toIntOrNull() ?: 10
        val drinkBreak = binding.switchDrinkBreak.isChecked

        // Save locally for the app to use
        val prefs = getSharedPreferences("MatchConfigPrefs", Context.MODE_PRIVATE)
        prefs.edit().apply {
            putInt("periods", periods)
            putInt("duration", duration)
            putInt("breakDuration", breakDuration)
            putBoolean("drinkBreak", drinkBreak)
            apply()
        }

        // Note: You will eventually beam these settings to the watch ViewModel here
        Toast.makeText(this, "Match rules saved", Toast.LENGTH_SHORT).show()
    }

package com.example.refereetimer.phone

import android.content.Context
import android.graphics.*
import com.example.refereetimer.shared.models.MatchEvent
import com.example.refereetimer.shared.models.MatchEvent.CardType
import com.example.refereetimer.shared.models.MatchEvent.EventType

class FieldDiagramRenderer(private val context: Context) {
    private val WIDTH = 1200; private val HEIGHT = 800; private val MARGIN = 60f
    private val fieldLeft = MARGIN; private val fieldTop = MARGIN
    private val fieldRight = WIDTH - MARGIN; private val fieldBottom = HEIGHT - MARGIN
    private val fieldWidth = fieldRight - fieldLeft; private val fieldHeight = fieldBottom - fieldTop
    private val PA_LENGTH = 16.5f / 105f; private val PA_WIDTH = 40.32f / 68f
    private val GA_LENGTH = 5.5f / 105f;  private val GA_WIDTH  = 18.32f / 68f
    private val PERIOD_COLORS = listOf(Color.argb(200,33,150,243), Color.argb(200,76,175,80), Color.argb(200,255,152,0), Color.argb(200,156,39,176))

    fun renderPeriod(period: Int, trackPoints: List<GpsReferenceService.GpsTrackPoint>, events: List<MatchEvent>): Bitmap {
        val bitmap = Bitmap.createBitmap(WIDTH, HEIGHT, Bitmap.Config.ARGB_8888); val canvas = Canvas(bitmap)
        drawBackground(canvas); drawFieldMarkings(canvas)
        drawHeatmap(canvas, trackPoints, PERIOD_COLORS[(period-1).coerceIn(0, PERIOD_COLORS.size-1)])
        drawEventPins(canvas, events); drawLabel(canvas, "Period $period"); return bitmap
    }

    fun renderFullMatch(trackPoints: List<GpsReferenceService.GpsTrackPoint>, events: List<MatchEvent>, totalPeriods: Int): Bitmap {
        val bitmap = Bitmap.createBitmap(WIDTH, HEIGHT, Bitmap.Config.ARGB_8888); val canvas = Canvas(bitmap)
        drawBackground(canvas); drawFieldMarkings(canvas)
        for (p in 1..totalPeriods) drawHeatmap(canvas, trackPoints.filter { it.period == p }, PERIOD_COLORS[(p-1).coerceIn(0, PERIOD_COLORS.size-1)])
        drawEventPins(canvas, events); drawLabel(canvas, "Full Match"); return bitmap
    }

    private fun drawBackground(canvas: Canvas) { canvas.drawColor(Color.parseColor("#1B5E20")) }

    private fun drawFieldMarkings(canvas: Canvas) {
        val lp = Paint().apply { color = Color.WHITE; strokeWidth = 2f; style = Paint.Style.STROKE; isAntiAlias = true }
        val sp = Paint().apply { color = Color.WHITE; style = Paint.Style.FILL; isAntiAlias = true }
        canvas.drawRect(fieldLeft, fieldTop, fieldRight, fieldBottom, lp)
        val midY = fieldTop + fieldHeight/2f; canvas.drawLine(fieldLeft, midY, fieldRight, midY, lp)
        val cR = (9.15f/105f) * fieldWidth; canvas.drawCircle(fieldLeft+fieldWidth/2f, fieldTop+fieldHeight/2f, cR, lp)
        canvas.drawCircle(fieldLeft+fieldWidth/2f, fieldTop+fieldHeight/2f, 5f, sp)
        canvas.drawRect(RectF(fieldLeft+fieldWidth*((1-PA_WIDTH)/2f), fieldTop, fieldLeft+fieldWidth*((1+PA_WIDTH)/2f), fieldTop+fieldHeight*PA_LENGTH), lp)
        canvas.drawRect(RectF(fieldLeft+fieldWidth*((1-PA_WIDTH)/2f), fieldBottom-fieldHeight*PA_LENGTH, fieldLeft+fieldWidth*((1+PA_WIDTH)/2f), fieldBottom), lp)
        canvas.drawRect(RectF(fieldLeft+fieldWidth*((1-GA_WIDTH)/2f), fieldTop, fieldLeft+fieldWidth*((1+GA_WIDTH)/2f), fieldTop+fieldHeight*GA_LENGTH), lp)
        canvas.drawRect(RectF(fieldLeft+fieldWidth*((1-GA_WIDTH)/2f), fieldBottom-fieldHeight*GA_LENGTH, fieldLeft+fieldWidth*((1+GA_WIDTH)/2f), fieldBottom), lp)
        val psDist = 11f/105f
        canvas.drawCircle(fieldLeft+fieldWidth/2f, fieldTop+fieldHeight*psDist, 4f, sp)
        canvas.drawCircle(fieldLeft+fieldWidth/2f, fieldBottom-fieldHeight*psDist, 4f, sp)
    }

    private fun drawHeatmap(canvas: Canvas, trackPoints: List<GpsReferenceService.GpsTrackPoint>, baseColor: Int) {
        if (trackPoints.isEmpty()) return
        val gridW = 50; val gridH = 33; val density = Array(gridW) { IntArray(gridH) }; var maxDensity = 1
        for (p in trackPoints) {

package com.example.refereetimer.phone

import android.content.Context
import android.graphics.Bitmap
import com.example.refereetimer.shared.models.LeagueType
import com.example.refereetimer.shared.models.Player
import com.google.mlkit.vision.common.InputImage
import com.google.mlkit.vision.text.TextRecognition
import com.google.mlkit.vision.text.latin.TextRecognizerOptions
import kotlinx.coroutines.tasks.await

class OcrProcessor(private val context: Context) {
    private val recognizer = TextRecognition.getClient(TextRecognizerOptions.DEFAULT_OPTIONS)
    private var currentLeague = LeagueType.SJSL

    data class OcrResult(val players: List<Player>, val coaches: List<String>, val teamName: String, val flaggedCount: Int, val confidence: Float)

    fun setLeague(league: LeagueType) { currentLeague = league }

    suspend fun processGameCard(bitmap: Bitmap, league: LeagueType): OcrResult {
        val image      = InputImage.fromBitmap(bitmap, 0)
        val visionText = recognizer.process(image).await()
        val fullText   = visionText.text
        return when (league) {
            LeagueType.SJSL, LeagueType.EDP, LeagueType.SJGSL -> parseGotSportCard(fullText)
            LeagueType.HIGH_SCHOOL -> parseHighSchoolCard(fullText)
            else -> parseGenericCard(fullText)
        }
    }

    private fun parseGotSportCard(fullText: String): OcrResult {
        val lines   = fullText.lines().map { it.trim() }.filter { it.isNotBlank() }
        val players = mutableListOf<Player>(); val coaches = mutableListOf<String>(); var teamName = ""
        for (line in lines) { if (line.startsWith("H ") || line.startsWith("HOME")) { teamName = extractTeamName(line); break } }
        for (line in lines) { if (line.contains("Coach", ignoreCase = true)) { val n = extractCoachName(line); if (n.isNotBlank()) coaches.add(n) } }
        for (line in lines) { if (isGotSportRosterRow(line)) parseGotSportRosterRow(line)?.let { players.add(it) } }
        val flagged = players.count { it.flagged }
        val confidence = if (players.isEmpty()) 0f else players.count { !it.flagged }.toFloat() / players.size
        return OcrResult(players, coaches, teamName, flagged, confidence)
    }

    private fun isGotSportRosterRow(line: String): Boolean {
        val parts = line.trim().split("\\s+".toRegex())
        if (parts.size < 3) return false
        return parts[0].toIntOrNull()?.let { it in 1..30 } == true && parts[1].length >= 5 && parts[1].all { it.isDigit() }
    }

    private fun parseGotSportRosterRow(line: String): Player? {
        val parts = line.trim().split("\\s+".toRegex()); if (parts.size < 4) return null
        return try {
            val jerseyStr = parts[2]; val jerseyNum = jerseyStr.filter { it.isDigit() }.toIntOrNull() ?: return null
            if (jerseyNum !in 1..99) return null
            val withoutDob = line.replace(Regex("""\d{2}/\d{2,4}$"""), "").trim()
            val partsNoDob = withoutDob.split("\\s+".toRegex())
            val name = if (partsNoDob.size > 3) partsNoDob.subList(3, partsNoDob.size).joinToString(" ").trim() else ""
            if (name.isBlank()) return null
            Player(jerseyNum, cleanName(name), jerseyStr.contains("/") || name.length < 3 || name.any { it.isDigit() })
        } catch (e: Exception) { null }
    }

    private fun parseHighSchoolCard(fullText: String): OcrResult {
        val lines   = fullText.lines().map { it.trim() }.filter { it.isNotBlank() }
        val players = mutableListOf<Player>(); val coaches = mutableListOf<String>()
        val isMapleShadeType = fullText.contains("Position", ignoreCase = true) && fullText.contains("Grade", ignoreCase = true)
        val teamName = lines.firstOrNull { it.contains("Soccer", ignoreCase = true) || it.contains("Varsity", ignoreCase = true) } ?: ""
        for (line in lines) { if (line.contains("Coach", ignoreCase = true) || line.contains("Head Coach", ignoreCase = true)) {
            val name = line.replace(Regex("(Head|Asst\\.?|Assistant)\\s+Coach\\s*[-–]?", RegexOption.IGNORE_CASE), "").trim()
            if (name.isNotBlank()) coaches.add(name) } }
        var tableStart = lines.indexOfFirst { it.contains("PLAYER", ignoreCase = true) || (it.contains("NAME", ignoreCase = true) && it.contains("NUMBER", ignoreCase = true)) }
        if (tableStart < 0) tableStart = 0
        for (i in tableStart + 1 until lines.size) {
            val line = lines[i]
            if (line.isBlank() || line.contains("Coach", ignoreCase = true) || line.contains("Date", ignoreCase = true)) continue
            val player = if (isMapleShadeType) parseMapleShadeRow(line) else parseMedfordTechRow(line)
            player?.let { players.add(it) }
        }
        val flagged = players.count { it.flagged }
        val confidence = if (players.isEmpty()) 0f else players.count { !it.flagged }.toFloat() / players.size
        return OcrResult(players, coaches, teamName, flagged, confidence)
    }

    private fun parseMedfordTechRow(line: String): Player? {
        val numbers = Regex("""\b(\d{1,2})\b""").findAll(line).map { it.value.toInt() }.filter { it in 1..99 }.toList()
        if (numbers.isEmpty()) return null
        val jerseyNum = numbers.firstOrNull() ?: return null
        val name = cleanName(line.replace(Regex("^[A-Z]\\s+"), "").replace(jerseyNum.toString(), "").replace(Regex("\\d+"), "").trim())
        if (name.isBlank() || name.length < 3) return null
        return Player(jerseyNum, name, name.any { it.isDigit() } || numbers.size > 2)
    }

    private fun parseMapleShadeRow(line: String): Player? {
        val parts = line.trim().split("\\s{2,}|\\t".toRegex())
        if (parts.size < 3) return null
        val name      = cleanName(parts[0]); if (name.isBlank()) return null
        val jerseyStr = parts.getOrNull(2) ?: return null
        val jerseyNum = jerseyStr.filter { it.isDigit() }.toIntOrNull() ?: return null
        if (jerseyNum !in 1..99) return null
        return Player(jerseyNum, name, jerseyStr.contains("/") || name.length < 3)
    }

    private fun parseGenericCard(fullText: String): OcrResult {
        val lines   = fullText.lines().map { it.trim() }.filter { it.isNotBlank() }
        val players = mutableListOf<Player>()
        for (line in lines) {
            val jerseyNum = Regex("""\b(\d{1,2})\b""").find(line)?.value?.toIntOrNull() ?: continue
            if (jerseyNum !in 1..99) continue
            val name = cleanName(line.replace(jerseyNum.toString(), "").replace(Regex("\\d+"), "").trim())
            if (name.length >= 3) players.add(Player(jerseyNum, name, true))
        }
        return OcrResult(players, emptyList(), "", players.size, 0.5f)
    }

    private fun extractTeamName(line: String) = line.replace(Regex("^[HA]\\s+"), "").replace(Regex("\\d{4}$"), "").trim()
    private fun extractCoachName(line: String) = line.replace(Regex("Coach\\s*\\([^)]*\\)", RegexOption.IGNORE_CASE), "").replace(Regex("\\d{7,}"), "").trim()
    private fun cleanName(raw: String) = raw.replace(Regex("[^A-Za-z\\s'\\-]"), "").replace(Regex("\\s+"), " ").trim()
        .split(" ").joinToString(" ") { it.lowercase().replaceFirstChar { c -> c.uppercase() } }
}

                    _timerDisplay.postValue(formatMillis(remainingMillis))
                }
            }

                player = Player(json.getInt("playerNumber"), json.getString("playerName")),
                minute = json.getInt("minute"),
                period = json.getInt("period"),
                elapsedMillis = json.getLong("elapsedMillis"),
                cardType = if (json.has("cardType")) MatchEvent.CardType.valueOf(json.getString("cardType")) else null,
                misconductCode = if (json.has("misconductCode")) json.getString("misconductCode") else null,

    private fun evaluateKftmStatus() {
        val kftmEvents = _events.value?.filter { it.type == MatchEvent.EventType.KFTM_KICK } ?: return
        if (kftmEvents.isEmpty()) return

        var homeKicks = 0; var awayKicks = 0
        var homeGoals = 0; var awayGoals = 0

        kftmEvents.forEach { e ->
            if (e.team.isHome) {
                homeKicks++
                if (e.kftmResult == MatchEvent.KftmResult.SCORED) homeGoals++
            } else {
                awayKicks++
                if (e.kftmResult == MatchEvent.KftmResult.SCORED) awayGoals++
            }
        }

        var winner: String? = null
        val homeRemainingBase = maxOf(0, 5 - homeKicks)
        val awayRemainingBase = maxOf(0, 5 - awayKicks)

        if (homeKicks <= 5 && awayKicks <= 5) {
            if (homeGoals > awayGoals + awayRemainingBase) winner = "HOME"
            else if (awayGoals > homeGoals + homeRemainingBase) winner = "AWAY"
        }
        else if (homeKicks == awayKicks) {
            if (homeGoals > awayGoals) winner = "HOME"
            else if (awayGoals > homeGoals) winner = "AWAY"
        }

        if (winner != null) {
            val finalScore = "$homeGoals - $awayGoals"
            _kftmWinner.postValue(Pair(winner, finalScore))
        }
    }

package com.example.refereetimer.phone

import android.app.Notification
import android.app.NotificationChannel
import android.app.NotificationManager
import android.app.Service
import android.content.Intent
import android.os.IBinder
import androidx.core.app.NotificationCompat
import com.google.android.gms.location.*
import com.google.android.gms.wearable.PutDataMapRequest
import com.google.android.gms.wearable.Wearable
import kotlinx.coroutines.*
import kotlinx.coroutines.tasks.await

class GpsReferenceService : Service() {

    private lateinit var fusedLocationClient: FusedLocationProviderClient
    private val serviceScope = CoroutineScope(Dispatchers.IO + Job())

    private var calibratedLat = 0.0; private var calibratedLng = 0.0
    private var isCalibrated  = false
    private var currentLat    = 0.0; private var currentLng    = 0.0
    private var currentAccuracy = 99f
    private var lastCorrectionSentAt = 0L
    private val CORRECTION_INTERVAL  = 30_000L

    val gpsTrack = mutableListOf<GpsTrackPoint>()

    data class GpsTrackPoint(
        val lat: Double, val lng: Double,
        val fieldX: Float, val fieldY: Float,
        val period: Int, val elapsed: Long,
        val timestamp: Long = System.currentTimeMillis()
    )

    private val locationCallback = object : LocationCallback() {
        override fun onLocationResult(result: LocationResult) {
            val loc = result.lastLocation ?: return
            currentLat = loc.latitude; currentLng = loc.longitude; currentAccuracy = loc.accuracy
            if (!isCalibrated) return
            val driftLat = currentLat - calibratedLat; val driftLng = currentLng - calibratedLng
            val driftM   = haversineMeters(calibratedLat, calibratedLng, currentLat, currentLng)
            val now      = System.currentTimeMillis()
            if (driftM > 0.5f && now - lastCorrectionSentAt > CORRECTION_INTERVAL) {
                sendDriftCorrection(driftLat, driftLng); lastCorrectionSentAt = now
            }
        }
    }

    override fun onCreate() {
        super.onCreate()
        fusedLocationClient = LocationServices.getFusedLocationProviderClient(this)
        startForeground(NOTIFICATION_ID, buildNotification())
        startLocationUpdates()
    }

    override fun onStartCommand(intent: Intent?, flags: Int, startId: Int): Int {
        when (intent?.action) {
            ACTION_SET_CALIBRATED_POSITION -> { calibratedLat = currentLat; calibratedLng = currentLng; isCalibrated = true }
            ACTION_ADD_TRACK_POINT -> gpsTrack.add(GpsTrackPoint(
                lat = intent.getDoubleExtra(EXTRA_LAT, 0.0), lng = intent.getDoubleExtra(EXTRA_LNG, 0.0),
                fieldX = intent.getFloatExtra(EXTRA_FIELD_X, 0f), fieldY = intent.getFloatExtra(EXTRA_FIELD_Y, 0f),
                period = intent.getIntExtra(EXTRA_PERIOD, 1),   elapsed = intent.getLongExtra(EXTRA_ELAPSED, 0L)))
            ACTION_STOP -> { stopLocationUpdates(); stopSelf() }
        }
        return START_STICKY
    }

    private fun startLocationUpdates() {
        try {
            val request = LocationRequest.Builder(Priority.PRIORITY_HIGH_ACCURACY, 5_000L)
                .setMinUpdateIntervalMillis(5_000L).setMaxUpdateDelayMillis(10_000L).build()
            fusedLocationClient.requestLocationUpdates(request, locationCallback, mainLooper)
        } catch (e: SecurityException) {}
    }

    private fun stopLocationUpdates() { fusedLocationClient.removeLocationUpdates(locationCallback) }

    private fun sendDriftCorrection(driftLat: Double, driftLng: Double) {
        serviceScope.launch {
            try {
                val request = PutDataMapRequest.create("/data/gps/batch").apply {
                    dataMap.putDouble("drift_lat", driftLat); dataMap.putDouble("drift_lng", driftLng)
                    dataMap.putFloat("accuracy", currentAccuracy); dataMap.putLong("timestamp", System.currentTimeMillis())
                }
                Wearable.getDataClient(this@GpsReferenceService).putDataItem(request.asPutDataRequest()).await()
            } catch (e: Exception) {}
        }
    }

    private fun buildNotification(): Notification {
        val channelId = "gps_reference"
        getSystemService(NotificationManager::class.java).createNotificationChannel(
            NotificationChannel(channelId, "GPS Reference Station", NotificationManager.IMPORTANCE_LOW))
        return NotificationCompat.Builder(this, channelId)
            .setContentTitle("Referee Timer").setContentText("GPS reference active")
            .setSmallIcon(android.R.drawable.ic_menu_mylocation).setOngoing(true).build()
    }

    private fun haversineMeters(lat1: Double, lng1: Double, lat2: Double, lng2: Double): Float {
        val dLat = Math.toRadians(lat2 - lat1); val dLng = Math.toRadians(lng2 - lng1)
        val a = Math.sin(dLat/2)*Math.sin(dLat/2) + Math.cos(Math.toRadians(lat1))*Math.cos(Math.toRadians(lat2))*Math.sin(dLng/2)*Math.sin(dLng/2)
        return (6_371_000.0 * 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1-a))).toFloat()
    }

    override fun onBind(intent: Intent?): IBinder? = null
    override fun onDestroy() { super.onDestroy(); stopLocationUpdates(); serviceScope.coroutineContext[Job]?.cancel() }

    companion object {
        const val NOTIFICATION_ID = 1001
        const val ACTION_SET_CALIBRATED_POSITION = "com.refereetimer.SET_CALIBRATED"
        const val ACTION_ADD_TRACK_POINT = "com.refereetimer.ADD_TRACK_POINT"
        const val ACTION_STOP = "com.refereetimer.GPS_STOP"
        const val EXTRA_LAT = "lat"; const val EXTRA_LNG = "lng"
        const val EXTRA_FIELD_X = "field_x"; const val EXTRA_FIELD_Y = "field_y"
        const val EXTRA_PERIOD = "period"; const val EXTRA_ELAPSED = "elapsed"
    }
}

    fun generate(): File {

    private fun addCoverPage(doc: PdfDocument, pageNum: Int): Int {
        val page = doc.startPage(PdfDocument.PageInfo.Builder(PAGE_W, PAGE_H, pageNum).create())
        val canvas = page.canvas

        canvas.drawColor(CLR_BG)
        canvas.drawRect(0f, 0f, PAGE_W.toFloat(), 120f, Paint().apply { color = CLR_HEADER })
        drawText(canvas, "MATCH REPORT", PAGE_W / 2f, 55f, 36f, Color.WHITE, true, Paint.Align.CENTER)
        drawText(canvas, "Official Match Record", PAGE_W / 2f, 90f, 14f, Color.parseColor("#A5D6A7"), false, Paint.Align.CENTER)

        // ── SCORING CALCULATION LOGIC ──
        val regularPeriods = config?.regularPeriodsCount ?: 2
        var homeReg = 0; var awayReg = 0
        var homeOt = 0;  var awayOt = 0
        var homeKftm = 0; var awayKftm = 0
        var totalKicks = 0

        events.forEach { e ->
            if (e.type == EventType.GOAL) {
                val isHomeScorer = if (e.isOwnGoal) !e.team.isHome else e.team.isHome
                if (e.period <= regularPeriods) {
                    if (isHomeScorer) homeReg++ else awayReg++
                } else {
                    if (isHomeScorer) homeOt++ else awayOt++
                }
            } else if (e.type == EventType.KFTM_KICK) {
                totalKicks++
                if (e.kftmResult == KftmResult.SCORED) {
                    if (e.team.isHome) homeKftm++ else awayKftm++
                }
            }
        }

        val homeTotalPlay = homeReg + homeOt
        val awayTotalPlay = awayReg + awayOt
        val wentToOt = homeOt > 0 || awayOt > 0 || events.any { it.period > regularPeriods }
        val wentToKftm = totalKicks > 0

        // Format strings with asterisk for OT/KFTM winners
        var homeScoreStr = homeTotalPlay.toString()
        var awayScoreStr = awayTotalPlay.toString()

        if (wentToOt && !wentToKftm && homeTotalPlay > awayTotalPlay) homeScoreStr += "*"
        if (wentToOt && !wentToKftm && awayTotalPlay > homeTotalPlay) awayScoreStr += "*"
        if (wentToKftm && homeKftm > awayKftm) homeScoreStr += "*"
        if (wentToKftm && awayKftm > homeKftm) awayScoreStr += "*"

        // Background box for score
        canvas.drawRoundRect(RectF(PAGE_W / 2f - 220f, 150f, PAGE_W / 2f + 220f, 320f), 16f, 16f, Paint().apply { color = Color.parseColor("#F5F5F5") })

        // Main Title Line: Home-Eagles(Score*) vs Away-Giants(score)
        val homeFullStr = "$homeName($homeScoreStr)"
        val awayFullStr = "$awayName($awayScoreStr)"
        drawText(canvas, homeFullStr, PAGE_W / 2f - 40f, 200f, 24f, CLR_HOME, true, Paint.Align.RIGHT)
        drawText(canvas, "vs", PAGE_W / 2f, 200f, 20f, CLR_TEXT, false, Paint.Align.CENTER)
        drawText(canvas, awayFullStr, PAGE_W / 2f + 40f, 200f, 24f, CLR_AWAY, true, Paint.Align.LEFT)

        // Sub-scores for OT and KFTM
        var currentY = 240f
        if (wentToOt) {
            drawText(canvas, "OT ($homeOt - $awayOt)", PAGE_W / 2f, currentY, 14f, CLR_SUBTEXT, true, Paint.Align.CENTER)
            currentY += 25f
        }
        if (wentToKftm) {
            drawText(canvas, "KFTM ($homeKftm - $awayKftm) | Total Kicks: $totalKicks", PAGE_W / 2f, currentY, 14f, CLR_SUBTEXT, true, Paint.Align.CENTER)
        }

        // Meta Data
        drawText(canvas, SimpleDateFormat("MMMM d, yyyy", Locale.US).format(Date()), PAGE_W / 2f, 345f, 14f, CLR_SUBTEXT, false, Paint.Align.CENTER)
        drawText(canvas, config?.league?.name ?: "League", PAGE_W / 2f, 370f, 14f, CLR_SUBTEXT, false, Paint.Align.CENTER)

        // Bottom Stats Row
        val fouls = events.count { it.type == EventType.FOUL }
        val yellow = events.count { it.type == EventType.CARD && it.cardType == CardType.YELLOW }
        val red = events.count { it.type == EventType.CARD && it.cardType == CardType.RED }
        val stoppageMins = totalStoppageMillis / 60000

        doc.finishPage(page)
        return pageNum + 1
    }

    private fun setupGestures() {
        gestureDetector = GestureDetector(this, object : GestureDetector.SimpleOnGestureListener() {
            override fun onFling(e1: MotionEvent?, e2: MotionEvent, velocityX: Float, velocityY: Float): Boolean {
                if (e1 == null || e2 == null) return false

                // Only allow quick events if the match is running or paused
                val state = timerViewModel.timerState.value
                if (state != TimerViewModel.TimerState.RUNNING && state != TimerViewModel.TimerState.PAUSED && state != TimerViewModel.TimerState.TIME_UP) {
                    return false
                }

                val diffX = e2.x - e1.x
                val diffY = e2.y - e1.y

                // Check if it's a horizontal swipe
                if (abs(diffX) > abs(diffY) && abs(diffX) > 100 && abs(velocityX) > 100) {
                    if (diffX > 0) {
                        // Swipe Right -> Home Team Event
                        triggerQuickEvent(isHome = true)
                    } else {
                        // Swipe Left -> Away Team Event
                        triggerQuickEvent(isHome = false)
                    }
                    return true
                }
                return false
            }
        })

    fun reset() {
        _homeScore.value      = 0
        _awayScore.value      = 0
        _events.value         = mutableListOf()
        _lastEvent.value      = null
        _pendingPKEvent.value = null
        yellowCardCount.clear()
        redCardPlayers.clear()
    }

    fun addGoal(
        isHome: Boolean, scorerNumber: Int, assistNumber: Int?,
        isOwnGoal: Boolean, isPK: Boolean,
        minute: Int, period: Int, elapsedMillis: Long,
        gpsLocation: GpsPoint? = null
    ): MatchEvent? {
        val scoringTeam = if (isOwnGoal) {
            if (isHome) _awayTeam.value else _homeTeam.value
        } else {
            if (isHome) _homeTeam.value else _awayTeam.value
        }
        val team   = scoringTeam ?: return null
        val scorer = team.getPlayer(scorerNumber) ?: Player(scorerNumber, "Unknown")
        val assist = assistNumber?.let { team.getPlayer(it) ?: Player(it, "Unknown") }

        val event = MatchEvent(
            type = EventType.GOAL, team = team, player = scorer,
            assistPlayer = assist, minute = minute, period = period,
            elapsedMillis = elapsedMillis, isOwnGoal = isOwnGoal,
            isPK = isPK, gpsLocation = gpsLocation
        )
        if (isOwnGoal) {
            if (isHome) _awayScore.value = (_awayScore.value ?: 0) + 1
            else        _homeScore.value = (_homeScore.value ?: 0) + 1
        } else {
            if (isHome) _homeScore.value = (_homeScore.value ?: 0) + 1
            else        _awayScore.value = (_awayScore.value ?: 0) + 1
        }
        addEvent(event)
        return event
    }

        isHome: Boolean, playerNumber: Int, foulType: FoulType,
        isPK: Boolean, minute: Int, period: Int, elapsedMillis: Long,
        gpsLocation: GpsPoint? = null

                resolvedCardType = CardType.RED
                result = CardResult.RED
            }
        }

        val event = MatchEvent(
            type = EventType.CARD, team = team, player = player,
            minute = minute, period = period, elapsedMillis = elapsedMillis,
            cardType = resolvedCardType, gpsLocation = gpsLocation
        )
        addEvent(event)
        return result

                        true

    private fun computeAffineTransform(
        src: List<Pair<Float, Float>>,
        dst: List<Pair<Float, Float>>
    ): FloatArray {
        val n = src.size
        var sumX = 0f; var sumY = 0f; var sumU = 0f; var sumV = 0f
        var sumXX = 0f; var sumXY = 0f; var sumYY = 0f
        var sumXU = 0f; var sumYU = 0f; var sumXV = 0f; var sumYV = 0f

        for (i in 0 until n) {
            val x = src[i].first;  val y = src[i].second
            val u = dst[i].first;  val v = dst[i].second
            sumX+=x; sumY+=y; sumU+=u; sumV+=v
            sumXX+=x*x; sumXY+=x*y; sumYY+=y*y
            sumXU+=x*u; sumYU+=y*u; sumXV+=x*v; sumYV+=y*v
        }

        val det = sumXX * sumYY - sumXY * sumXY
        val a = if (det != 0f) (sumXU*sumYY - sumYU*sumXY) / det else 1f
        val b = if (det != 0f) (sumXX*sumYU - sumXY*sumXU) / det else 0f
        val c = (sumU - a*sumX - b*sumY) / n
        val d = if (det != 0f) (sumXV*sumYY - sumYV*sumXY) / det else 0f
        val e = if (det != 0f) (sumXX*sumYV - sumXY*sumXV) / det else 1f
        val f = (sumV - d*sumX - e*sumY) / n

        return floatArrayOf(a, b, c, d, e, f)
    }

            )
        } else null

            try {

                    DataLayerPaths.MSG_CALIBRATE_DONE

                    .putDataItem(request.asPutDataRequest().setUrgent())
                    .await()
            } catch (e: Exception) { /* Phone unavailable */ }
        }

        tts.speak(
            "Calibration complete. ${calibratedPoints.size} points captured.",
            TextToSpeech.QUEUE_FLUSH, null, null

            VibrationEffect.createWaveform(

        })
        finish()
    }

                    )

        }
    }

package com.example.refereetimer

import android.app.Application
import android.os.CountDownTimer
import android.os.Handler
import android.os.Looper
import androidx.lifecycle.AndroidViewModel

class TimerViewModel(application: Application) : AndroidViewModel(application) {

    enum class TimerState { READY, IDLE, RUNNING, PAUSED, TIME_UP, BREAK, MATCH_OVER }

    private val _timerState = MutableLiveData(TimerState.READY)
    val timerState: LiveData<TimerState> = _timerState

    private val _mainTimer = MutableLiveData(0L)
    val mainTimer: LiveData<Long> = _mainTimer

    private val _elapsedTimer = MutableLiveData(0L)
    val elapsedTimer: LiveData<Long> = _elapsedTimer

    private val _stoppageTimer = MutableLiveData(0L)
    val stoppageTimer: LiveData<Long> = _stoppageTimer

    private val _currentPeriod = MutableLiveData(1)
    val currentPeriod: LiveData<Int> = _currentPeriod

    private val _totalPeriods = MutableLiveData(3)
    val totalPeriods: LiveData<Int> = _totalPeriods

    private val _homeTeamName = MutableLiveData("")
    val homeTeamName: LiveData<String> = _homeTeamName

    private val _awayTeamName = MutableLiveData("")
    val awayTeamName: LiveData<String> = _awayTeamName

    var regularPeriodsCount = 2
    private var regularDurationMillis = 45 * 60 * 1000L
    private var otDurationMillis = 10 * 60 * 1000L
    private var breakDurationMillis = 15 * 60 * 1000L
    private var remainingMainTimeMillis = 45 * 60 * 1000L

    private var mainCountdownTimer: CountDownTimer? = null
    private val elapsedHandler = Handler(Looper.getMainLooper())
    private val stoppageHandler = Handler(Looper.getMainLooper())

    private val elapsedRunnable = object : Runnable {
        override fun run() {
            _elapsedTimer.value = (_elapsedTimer.value ?: 0) + 1000L
            elapsedHandler.postDelayed(this, 1000L)
        }
    }

    private val stoppageRunnable = object : Runnable {
        override fun run() {
            _stoppageTimer.value = (_stoppageTimer.value ?: 0) + 1000L
            stoppageHandler.postDelayed(this, 1000L)
        }
    }

    fun setMatchParameters(
        regMins: Int, regSecs: Int, periods: Int,
        breakMins: Int, isOT: Boolean, otMins: Int, otSecs: Int,
        homeName: String, awayName: String
    ) {
        regularDurationMillis = (regMins * 60 * 1000L) + (regSecs * 1000L)
        breakDurationMillis   = breakMins * 60 * 1000L
        regularPeriodsCount   = periods
        _totalPeriods.value   = if (isOT) regularPeriodsCount + 2 else regularPeriodsCount
        otDurationMillis      = (otMins * 60 * 1000L) + (otSecs * 1000L)
        _homeTeamName.value   = homeName
        _awayTeamName.value   = awayName
        resetMatch()
    }

    fun resetMatch() {
        mainCountdownTimer?.cancel()
        elapsedHandler.removeCallbacks(elapsedRunnable)
        stoppageHandler.removeCallbacks(stoppageRunnable)
        _currentPeriod.value        = 1
        remainingMainTimeMillis     = regularDurationMillis
        _mainTimer.value            = regularDurationMillis
        _elapsedTimer.value         = 0L
        _stoppageTimer.value        = 0L
        _timerState.value           = TimerState.READY
    }

    fun onMainButtonClick() {
        when (_timerState.value) {
            TimerState.READY -> { _timerState.value = TimerState.IDLE }
            TimerState.IDLE -> startTimers()
            TimerState.RUNNING -> pauseTimers()
            TimerState.PAUSED -> resumeTimers()
            TimerState.TIME_UP -> endPeriod()
            TimerState.BREAK -> startNextPeriod()
            else -> {}
        }
        syncWithPhone()
    }

    private fun startTimers() {
        stoppageHandler.removeCallbacks(stoppageRunnable)
        mainCountdownTimer = object : CountDownTimer(remainingMainTimeMillis, 1000) {
            override fun onTick(millisUntilFinished: Long) {
                _mainTimer.value = millisUntilFinished
                remainingMainTimeMillis = millisUntilFinished
            }
            override fun onFinish() {
                _mainTimer.value = 0L
                _timerState.value = TimerState.TIME_UP
                syncWithPhone()
            }
        }.start()

        if (_timerState.value != TimerState.RUNNING) {
            elapsedHandler.post(elapsedRunnable)
        }
        _timerState.value = TimerState.RUNNING
    }

    private fun pauseTimers() {
        mainCountdownTimer?.cancel()
        stoppageHandler.post(stoppageRunnable)
        _timerState.value = TimerState.PAUSED
    }

    private fun resumeTimers() {
        stoppageHandler.removeCallbacks(stoppageRunnable)
        startTimers()
    }

    private fun endPeriod() {
        elapsedHandler.removeCallbacks(elapsedRunnable)
        stoppageHandler.removeCallbacks(stoppageRunnable)

        if ((_currentPeriod.value ?: 1) < regularPeriodsCount) {
            _timerState.value = TimerState.BREAK
            _mainTimer.value = breakDurationMillis
            mainCountdownTimer = object : CountDownTimer(breakDurationMillis, 1000) {
                override fun onTick(ms: Long) { _mainTimer.value = ms }
                override fun onFinish() { _mainTimer.value = 0L }
            }.start()
        } else {
            _timerState.value = TimerState.MATCH_OVER
        }
    }

    private fun startNextPeriod() {
        mainCountdownTimer?.cancel()
        _currentPeriod.value = (_currentPeriod.value ?: 1) + 1
        _stoppageTimer.value = 0L
        remainingMainTimeMillis = if ((_currentPeriod.value ?: 1) > regularPeriodsCount) otDurationMillis else regularDurationMillis
        startTimers()
    }

    fun syncWithPhone() {
        val state = _timerState.value?.name ?: "IDLE"
        val message = "STATE=$state|TIME=$remainingMainTimeMillis"
        CoroutineScope(Dispatchers.IO).launch {
            try {
                val nodes = Wearable.getNodeClient(getApplication<Application>()).connectedNodes.await()
                for (node in nodes) {
                    Wearable.getMessageClient(getApplication<Application>())
                        .sendMessage(node.id, DataLayerPaths.PATH_SYNC_TIME_UPDATE, message.toByteArray())
                        .await()
                }
            } catch (e: Exception) {
                android.util.Log.e("SYNC_ERROR", "Failed to sync with phone", e)
            }
        }
    }
